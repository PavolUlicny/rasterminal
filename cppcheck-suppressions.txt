unusedFunction:src/framebuffer.h

// main.cpp carries one inline `cppcheck-suppress knownConditionTrueFalse` (the VT-gate
// condition, constant only because <windows.h> is absent on the Linux runner so the _WIN32
// body can't be analysed). Under --inline-suppr that inline directive is a two-sided
// constraint: if a future cppcheck version stops emitting the finding, the now-unmatched
// inline suppression would itself fail CI. Let it age out harmlessly instead. Scoped to the
// file, which holds no other suppression, so this masks nothing else.
unmatchedSuppression:src/main.cpp

// framebuffer.cpp carries two inline knownConditionTrueFalse suppressions,
// in transmit_shm and present_kitty: under _WIN32 the shm create is the
// nullptr stub, so their shm-failure conditions read as constant to the
// multi-config scan. Same aging-out rationale as the main.cpp entry above.
unmatchedSuppression:src/framebuffer.cpp

// platform.h's two inline knownConditionTrueFalse suppressions (both in the
// graphics query's shm-probe block, whose create is the nullptr stub in the
// _WIN32 configuration) need no unmatchedSuppression pairing: cppcheck
// reports unmatched suppressions for
// primary inputs only, and a header is never one (verified empirically on
// 2.13.0 with the CI-verbatim command; an earlier claim here that the
// unmatched half silently set the exit code did not reproduce).

// Raw loops kept where an STL-algorithm rewrite would need a lambda and read worse. The
// platform.h entry is a separate case (a constexpr context); its reason lives at the code site.
useStlAlgorithm:src/mesh.cpp
useStlAlgorithm:src/mesh_obj.cpp
useStlAlgorithm:src/mesh_ply.cpp
useStlAlgorithm:src/platform.h
useStlAlgorithm:src/renderer.cpp
